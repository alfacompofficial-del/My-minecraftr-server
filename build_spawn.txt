# === ВОРОТА СПАВНА ===
# (Строй вручную для красоты)

# === УСТАНОВИТЬ СПАВН ===
setspawn

# === СОЗДАТЬ РЕГИОН СПАВНА (WorldGuard) ===
# 1. Встань на угол спавна
# 2. //wand (получи топорик WorldEdit)
# 3. Выдели регион: левый клик + правый клик
//pos1 -20,50,-20
//pos2 20,100,20
/region define spawn
/region flag spawn build deny
/region flag spawn pvp deny
/region flag spawn mob-spawning deny

echo "Спавн создан!"
