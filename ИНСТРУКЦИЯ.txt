# ============ ШАГ 4: НАСТРОЙКА СПАВНА ============
# Зайди на сервер как Bilol29146192
# Команды для создания спавна:
#   /gamemode creative    - переключиться в творческий режим
#   /tp 0 65 0           - телепорт на центр спавна
# Построй спавн вручную (или используй WorldEdit)
# Затем:
#   /setspawn            - установить точку спавна
#   /authspawn           - установить точку для неавторизованных (y=255)
#   /tp 0 255 0
#   /authspawn

# ============ ШАГ 5: ЗАЩИТА СПАВНА (WorldGuard) ============

# Выдели регион вокруг спавна левым и правым кликом деревянным топором
# Затем:
#   /region define spawn
#   /region flag spawn build deny
#   /region flag spawn pvp deny
#   /region flag spawn mob-spawning deny
#   /region flag spawn entry allow

# ============ КОМАНДЫ ИГРОКА ============
# /register <пароль> <пароль>  - регистрация
# /login <пароль>              - вход
# /kit start                   - стартовый набор (1 раз)
# /sethome [имя]               - установить дом
# /home [имя]                  - телепорт домой
# /spawn                       - телепорт на спавн
# /tpa <ник>                   - запрос телепорта
# /balance или /bal            - баланс
# /pay <ник> <сумма>           - перевести деньги
# /ah                          - аукционный дом
# /ah sell <цена>              - продать предмет в руке
# /msg <ник> <сообщение>       - личное сообщение

# ============ КОМАНДЫ АДМИНИСТРАТОРА ============
# Все команды обычного игрока +
# /op <ник>                    - дать OP
# /ban <ник> [причина]         - забанить
# /unban <ник>                 - разбанить  
# /kick <ник> [причина]        - кикнуть
# /gamemode <режим> [ник]      - сменить режим игры
# //wand                       - инструмент WorldEdit
# //set <блок>                 - заполнить блоками
# //copy, //paste              - копировать/вставить
# /region define <имя>         - создать регион WorldGuard
# /lp user <ник> parent set admin  - назначить администратора
